Create Table Клиент (
	ID BIGINT Primary Key Identity(1,1),
	ФИО Nvarchar(255) Not Null,
	Паспорт Bigint Not Null Unique,
	Адрес Nvarchar(255),
	Телефон Bigint,
	Кол_во_купленых_туров Int Default 0
);

Create Table Транспорт (
    ID BIGINT Primary Key Identity(1,1),
    Название Nvarchar(255) Not Null,
);

Create Table Отель (
    ID BIGINT Primary Key Identity(1,1),
    Название Nvarchar(255) Not Null,
    Кол_во_звезд Int Check (Кол_во_звезд Between 1 and 5), 
    Стоимость_за_день Int Check (Стоимость_за_день > 0)
);

Create Table Тур (
    ID BIGINT Primary Key Identity(1,1),
    Страна Nvarchar(100) Not Null,
    Регион Nvarchar(100),
    Программа Nvarchar(MAX),
    Длительность_Дней Int Check (Длительность_Дней > 0)
);

Create Table Транспорт_Тур (
    Тур_ID Bigint Not Null,
    Транспорт_ID Bigint Not Null,
    Primary Key (Тур_ID, Транспорт_ID),
    Foreign Key (Тур_ID) References Тур(ID) On Delete Cascade,
    Foreign Key (Транспорт_ID) References Транспорт(ID) On Delete Cascade
);

Create Table Бронирование (
    ID BIGINT Primary Key Identity(1,1),
    Дата_начала Date Not Null,
    Дата_окончания Date Not Null,
    Кол_во_человек Int Check (Кол_во_человек > 0),
    Клиент_ID BIGINT Not Null,
    Тур_ID BIGINT Not Null,
    Отель_ID BIGINT Not Null,
    Foreign Key (Клиент_ID) References Клиент(ID),
    Foreign Key (Тур_ID) References Тур(ID),
    Foreign Key (Отель_ID) References Отель(ID),
    Check (Дата_окончания >= Дата_начала) 
);

Create Table Претензия (
    ID BIGINT Primary Key Identity(1,1),
    Текст Nvarchar(MAX) Not Null,
    Клиент_ID BIGINT Not Null,
    Бронирование_ID BIGINT Not Null,
    Foreign Key (Клиент_ID) References Клиент(ID),
    Foreign Key (Бронирование_ID) References Бронирование(ID)
);
